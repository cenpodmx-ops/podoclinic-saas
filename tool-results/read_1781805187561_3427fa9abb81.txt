   182→function EquipoTab() {
   183→  const qc = useQueryClient()
   184→  const [open, setOpen] = useState(false)
   185→  const [editing, setEditing] = useState<any>(null)
   186→
   187→  const { data, isLoading } = useQuery({
   188→    queryKey: ['podologos'],
   189→    queryFn: () => fetch('/api/podologos?includeInactive=1').then((r) => r.json()),
   190→  })
   191→
   192→  const save = useMutation({
   193→    mutationFn: async (body: any) => {
   194→      if (editing) {
   195→        const res = await fetch(`/api/podologos/${editing.id}`, {
   196→          method: 'PATCH',
   197→          headers: { 'Content-Type': 'application/json' },
   198→          body: JSON.stringify(body),
   199→        })
   200→        if (!res.ok) throw new Error('Error')
   201→        return res.json()
   202→      } else {
   203→        const res = await fetch('/api/podologos', {
   204→          method: 'POST',
   205→          headers: { 'Content-Type': 'application/json' },
   206→          body: JSON.stringify(body),
   207→        })
   208→        if (!res.ok) throw new Error('Error')
   209→        return res.json()
   210→      }
   211→    },
   212→    onSuccess: () => {
   213→      toast.success(editing ? 'Podólogo actualizado' : 'Podólogo creado')
   214→      qc.invalidateQueries({ queryKey: ['podologos'] })
   215→      setOpen(false)
   216→      setEditing(null)
   217→    },
   218→    onError: () => toast.error('Error al guardar'),
   219→  })
   220→
   221→  const deactivate = useMutation({
   222→    mutationFn: (id: string) => fetch(`/api/podologos/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   223→    onSuccess: () => {
   224→      toast.success('Podólogo desactivado')
   225→      qc.invalidateQueries({ queryKey: ['podologos'] })
   226→    },
   227→  })
   228→
   229→  const rows: any[] = Array.isArray(data) ? data : (data?.rows || [])
   230→
   231→  return (
   232→    <Card className="shadow-sm">
   233→      <CardHeader className="flex flex-row items-center justify-between">
   234→        <CardTitle className="text-base">Equipo de podólogos</CardTitle>
   235→        <Button size="sm" onClick={() => { setEditing(null); setOpen(true) }} style={{ backgroundColor: '#0a3143' }}>
   236→          <Plus className="h-4 w-4 mr-1" /> Nuevo
   237→        </Button>
   238→      </CardHeader>
   239→      <CardContent className="p-0">
   240→        {isLoading ? (
   241→          <div className="p-4 space-y-2">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-12" />)}</div>
   242→        ) : rows.length === 0 ? (
   243→          <div className="p-12 text-center text-muted-foreground">Sin podólogos registrados</div>
   244→        ) : (
   245→          <div className="overflow-x-auto">
   246→            <Table>
   247→              <TableHeader>
   248→                <TableRow>
   249→                  <TableHead>Nombre</TableHead>
   250→                  <TableHead>Especialidad</TableHead>
   251→                  <TableHead>Cédula</TableHead>
   252→                  <TableHead>Teléfono</TableHead>
   253→                  <TableHead>Comisión</TableHead>
   254→                  <TableHead>Meta mensual</TableHead>
   255→                  <TableHead>Estado</TableHead>
   256→                  <TableHead className="text-right">Acciones</TableHead>
   257→                </TableRow>
   258→              </TableHeader>

   260→                {rows.map((p: any) => (
   261→                  <TableRow key={p.id}>
   262→                    <TableCell className="font-medium">{p.name}</TableCell>
   263→                    <TableCell className="text-sm">{p.specialty || '—'}</TableCell>
   264→                    <TableCell className="text-sm">{p.cedula || '—'}</TableCell>
   265→                    <TableCell className="text-sm">{p.phone || '—'}</TableCell>
   266→                    <TableCell><Badge variant="outline">{p.commissionPct}%</Badge></TableCell>
   267→                    <TableCell className="text-sm">
   268→                      {p.monthlyGoalConsults ? `${p.monthlyGoalConsults} citas` : '—'}
   269→                      {p.monthlyGoalRevenue ? ` / ${fmtMoney(p.monthlyGoalRevenue)}` : ''}
   270→                    </TableCell>
   271→                    <TableCell>
   272→                      {p.active ? <Badge className="bg-emerald-100 text-emerald-700">Activo</Badge> : <Badge variant="secondary">Inactivo</Badge>}

   275→                      <Button variant="ghost" size="icon" onClick={() => { setEditing(p); setOpen(true) }}>
   276→                        <Pencil className="h-4 w-4" />

   279→                        <Button variant="ghost" size="icon" className="text-red-600" onClick={() => deactivate.mutate(p.id)}>
   280→                          <Plus className="h-4 w-4 rotate-45" />

   284→                  </TableRow>

   287→            </Table>

   456→function FacturacionTab() {
   457→  const qc = useQueryClient()
   458→  const [loading, setLoading] = useState(false)
   459→  const [mensaje, setMensaje] = useState<{ tipo: 'exito' | 'error'; texto: string } | null>(null)
   460→
   461→  const { data: faStatus, isLoading } = useQuery({
   462→    queryKey: ['facturapi-status'],
   463→    queryFn: () => fetch('/api/config/facturapi').then((r) => r.json()),
   464→  })
   465→
   466→  async function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
   467→    e.preventDefault()
   468→    setLoading(true)
   469→    setMensaje(null)
   470→    const formData = new FormData(e.currentTarget)
   471→    try {
   472→      const res = await fetch('/api/config/facturapi', { method: 'POST', body: formData })
   473→      const j = await res.json()
   474→      if (!res.ok) throw new Error(j?.error || 'Error')
   475→      setMensaje({ tipo: 'exito', texto: j.message || 'Configuración guardada correctamente.' })
   476→      qc.invalidateQueries({ queryKey: ['facturapi-status'] })
   477→      qc.invalidateQueries({ queryKey: ['config'] })
   478→      // Limpiar el form
   479→      ;(e.target as HTMLFormElement).reset()
   480→    } catch (err: any) {
   481→      setMensaje({ tipo: 'error', texto: err.message || 'Error' })
   482→    } finally {
   483→      setLoading(false)
   484→    }
