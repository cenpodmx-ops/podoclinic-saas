    92→export function PrescriptionEditor() {
    93→  const qc = useQueryClient()
    94→  const fileInputRef = useRef<HTMLInputElement>(null)
    95→
    96→  // Cargar configuración existente
    97→  const { data: configData, isLoading } = useQuery<any>({
    98→    queryKey: ['config'],
    99→    queryFn: () => fetch('/api/config').then((r) => r.json()),
   100→  })
   101→
   102→  const clinic = configData?.clinic
   103→
   104→  // Estado del diseño. Se inicializa una sola vez cuando llega la data.
   105→  const [design, setDesign] = useState<PrescriptionDesign | null>(null)
   106→  const [uploadedLogoUrl, setUploadedLogoUrl] = useState<string | null>(null)
   107→
   108→  // Inicializar diseño desde config (una sola vez)
   109→  if (!isLoading && configData && !design) {
   110→    let parsed: PrescriptionDesign = DEFAULT_DESIGN
   111→    const raw = configData?.config?.prescriptionDesign
   112→    if (raw) {
   113→      try {
   114→        parsed = { ...DEFAULT_DESIGN, ...(JSON.parse(raw) as Partial<PrescriptionDesign>) }
   115→      } catch {
   116→        parsed = DEFAULT_DESIGN
   117→      }
   118→    }
   119→    // Si la clínica ya tiene logo, mostrarlo por defecto (logoUrl='auto')
   120→    if (clinic?.logoUrl && parsed.logoUrl === undefined) {
   121→      parsed.logoUrl = 'auto'
   122→    }
   123→    setDesign(parsed)
   124→  }
   125→
   126→  function update(patch: Partial<PrescriptionDesign>) {
   127→    setDesign((d) => (d ? { ...d, ...patch } : d))
   128→  }
   129→
   130→  // Guardar diseño
   131→  const saveMut = useMutation({
   132→    mutationFn: async (body: PrescriptionDesign) => {
   133→      const r = await fetch('/api/config/plantillas', {
   134→        method: 'PATCH',
   135→        headers: { 'Content-Type': 'application/json' },
   136→        body: JSON.stringify({ prescriptionDesign: JSON.stringify(body) }),
   137→      })
   138→      if (!r.ok) {
   139→        const e = await r.json().catch(() => ({}))
   140→        throw new Error(e.error || 'Error al guardar')
   141→      }
   142→      return r.json()
   143→    },
   144→    onSuccess: () => {
   145→      toast.success('Diseño de receta guardado')
   146→      qc.invalidateQueries({ queryKey: ['config'] })
   147→    },
   148→    onError: (e: Error) => toast.error(e.message),
   149→  })
   150→
   151→  // Subir logo
   152→  const uploadMut = useMutation({
   153→    mutationFn: async (file: File) => {
   154→      const fd = new FormData()
   155→      fd.append('file', file)
   156→      const r = await fetch('/api/config/logo', { method: 'POST', body: fd })
   157→      if (!r.ok) {
   158→        const e = await r.json().catch(() => ({}))
   159→        throw new Error(e.error || 'Error al subir el logo')
   160→      }
   161→      return r.json() as Promise<{ url: string }>
   162→    },
   163→    onSuccess: (data) => {
   164→      toast.success('Logo subido correctamente')
   165→      setUploadedLogoUrl(data.url)
   166→      update({ logoUrl: data.url })
   167→      qc.invalidateQueries({ queryKey: ['config'] })
   168→    },
   169→    onError: (e: Error) => toast.error(e.message),
   170→  })
   171→
   172→  function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
   173→    const f = e.target.files?.[0]
   174→    if (!f) return
   175→    uploadMut.mutate(f)
   176→  }
   177→
   178→  function handleReset() {
   179→    if (!confirm('¿Restablecer el diseño a los valores por defecto? Se perderán los cambios no guardados.')) return
   180→    setDesign({ ...DEFAULT_DESIGN, logoUrl: clinic?.logoUrl ? 'auto' : 'none' })
   181→    toast.info('Diseño restablecido. Recuerda guardar.')
   182→  }
   183→
   184→  function handlePrintTest() {
   185→    // Abre una ventana con una receta de muestra usando el diseño actual.
   186→    // Usamos una URL temporal que escribe el diseño en localStorage y luego
   187→    // la página de print-test lo lee. Pero como solo tenemos /api/recetas/[id]/print,
   188→    // usamos una alternativa: abrimos una ventana nueva y escribimos HTML inline
   189→    // que simula la receta con el diseño actual.
   190→    if (!design) return
   191→    const html = buildTestPrintHtml(design, {
   192→      ...SAMPLE_DATA,
   193→      clinic: clinic || null,
   194→    })
   195→    const w = window.open('', '_blank', 'width=900,height=1000')
   196→    if (!w) {
   197→      toast.error('Habilita las ventanas emergentes para imprimir la prueba.')
   198→      return
   199→    }
   200→    w.document.open()
   201→    w.document.write(html)
   202→    w.document.close()
   203→    setTimeout(() => {
   204→      try {
   205→        w.focus()
   206→        w.print()
   207→      } catch {}
   208→    }, 400)
   209→  }
   210→
   211→  if (isLoading || !design) {
   212→    return <Skeleton className="h-96" />
   213→  }
   214→
   215→  return (
   216→    <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   217→      {/* Columna izquierda: controles */}
   218→      <div className="space-y-4">
   219→        {/* Papel y logo */}
   220→        <Card>
   221→          <CardHeader>
   222→            <CardTitle className="text-base flex items-center gap-2">
   223→              <ImageIcon className="h-4 w-4 text-primary" /> Papel y logo
   224→            </CardTitle>
   225→          </CardHeader>
   226→          <CardContent className="space-y-4">
   227→            {/* Logo upload */}
   228→            <div className="space-y-2">
   229→              <Label className="text-xs">Logo de la clínica</Label>
   230→              <div className="flex items-center gap-3 flex-wrap">
   231→                <input
   232→                  ref={fileInputRef}
   233→                  type="file"
   234→                  accept="image/png,image/jpeg,image/webp,image/svg+xml"
   235→                  onChange={handleFileChange}
   236→                  className="hidden"
   237→                />
   238→                <Button
   239→                  type="button"
   240→                  variant="outline"
   241→                  size="sm"
   242→                  onClick={() => fileInputRef.current?.click()}
   243→                  disabled={uploadMut.isPending}
   244→                >
   245→                  <Upload className="h-3.5 w-3.5" />
   246→                  {uploadMut.isPending ? 'Subiendo…' : 'Subir logo'}
   247→                </Button>
   248→                {clinic?.logoUrl && (
   249→                  <Badge className="bg-emerald-100 text-emerald-700 text-xs">✓ Logo subido</Badge>
   250→                )}
   251→              </div>
   252→              {uploadedLogoUrl && (
   253→                <Badge className="bg-emerald-100 text-emerald-700 text-xs">Logo nuevo subido ✓</Badge>
   254→              )}
   255→            </div>
   256→
   257→            {/* Logo source */}
   258→            <div className="grid grid-cols-2 gap-3">
   259→              <div>
   260→                <Label className="text-xs">Fuente del logo</Label>
   261→                <Select
   262→                  value={design.logoUrl === 'none' ? 'none' : design.logoUrl === 'auto' || !design.logoUrl ? 'auto' : 'custom'}
   263→                  onValueChange={(v) => {
   264→                    if (v === 'none') update({ logoUrl: 'none' })
   265→                    else if (v === 'auto') update({ logoUrl: 'auto' })
   266→                    else if (uploadedLogoUrl) update({ logoUrl: uploadedLogoUrl })
   267→                    else update({ logoUrl: clinic?.logoUrl || 'auto' })
   268→                  }}
   269→                >
   270→                  <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   271→                  <SelectContent>
   272→                    <SelectItem value="auto">Automático (logo de la clínica)</SelectItem>
   273→                    <SelectItem value="custom">Logo subido</SelectItem>
   274→                    <SelectItem value="none">Sin logo</SelectItem>
   275→                  </SelectContent>
   276→                </Select>
   277→              </div>
   278→              <div>
   279→                <Label className="text-xs">Posición del logo</Label>
   280→                <Select value={design.logoPosition || 'left'} onValueChange={(v) => update({ logoPosition: v as any })}>
   281→                  <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   282→                  <SelectContent>
   283→                    <SelectItem value="left">Izquierda</SelectItem>
   284→                    <SelectItem value="center">Centro</SelectItem>
   285→                    <SelectItem value="right">Derecha</SelectItem>
   286→                  </SelectContent>
   287→                </Select>
   288→              </div>
   289→            </div>
   290→
   291→            <SliderControl
   292→              label="Tamaño del logo"
   293→              value={design.logoSize ?? 78}
   294→              min={40}
   295→              max={400}
   296→              unit="px"
   297→              onChange={(v) => update({ logoSize: v })}
   298→            />
   299→            <SliderControl
   300→              label="Opacidad del logo"
   301→              value={design.logoOpacity ?? 100}
   302→              min={10}
   303→              max={100}
   304→              unit="%"
   305→              onChange={(v) => update({ logoOpacity: v })}
   306→            />
   307→
   308→            <Separator />
   309→
   310→            {/* Watermark */}
   311→            <div className="space-y-3">
   312→              <div className="flex items-center justify-between">
   313→                <Label className="text-xs">Marca de agua (watermark)</Label>
   314→                <Switch
   315→                  checked={design.watermarkEnabled === true}
   316→                  onCheckedChange={(v) => update({ watermarkEnabled: v })}
   317→                />
   318→              </div>
   319→              {design.watermarkEnabled && (
   320→                <>
   321→                  <SliderControl
   322→                    label="Tamaño watermark"
   323→                    value={design.watermarkSize ?? 60}
   324→                    min={20}
   325→                    max={100}
   326→                    unit="%"
   327→                    onChange={(v) => update({ watermarkSize: v })}
   328→                  />
   329→                  <SliderControl
   330→                    label="Opacidad watermark"
   331→                    value={design.watermarkOpacity ?? 10}
   332→                    min={5}
   333→                    max={30}
   334→                    unit="%"
   335→                    onChange={(v) => update({ watermarkOpacity: v })}
   336→                  />
   337→                  <div>
   338→                    <Label className="text-xs">Posición watermark</Label>
   339→                    <Select
   340→                      value={design.watermarkPosition || 'center'}
   341→                      onValueChange={(v) => update({ watermarkPosition: v as any })}
   342→                    >
   343→                      <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   344→                      <SelectContent>
   345→                        <SelectItem value="center">Centro</SelectItem>
   346→                        <SelectItem value="top-right">Superior derecha</SelectItem>
   347→                        <SelectItem value="bottom-right">Inferior derecha</SelectItem>
   348→                      </SelectContent>
   349→                    </Select>
   350→                  </div>
   351→                </>
   352→              )}
   353→            </div>
   354→
   355→            <Separator />
   356→
   357→            {/* Paper size */}
   358→            <div>
   359→              <Label className="text-xs">Tamaño de papel</Label>
   360→              <Select value={design.paperSize || 'A4'} onValueChange={(v) => update({ paperSize: v as any })}>
   361→                <SelectTrigger><SelectValue /></SelectTrigger>
   362→                <SelectContent>
   363→                  <SelectItem value="MediaCarta">Media carta (140 × 216 mm)</SelectItem>
   364→                  <SelectItem value="Carta">Carta (216 × 279 mm)</SelectItem>
   365→                  <SelectItem value="A4">A4 (210 × 297 mm)</SelectItem>
   366→                </SelectContent>
   367→              </Select>
   368→            </div>
   369→
   370→            <SliderControl
   371→              label="Márgenes"
   372→              value={design.margins ?? 16}
   373→              min={10}
   374→              max={40}
   375→              unit="mm"
   376→              onChange={(v) => update({ margins: v })}
   377→            />
   378→          </CardContent>
   379→        </Card>
   380→
   381→        {/* Colores */}
   382→        <Card>
   383→          <CardHeader>
   384→            <CardTitle className="text-base">Colores</CardTitle>
   385→          </CardHeader>
   386→          <CardContent className="space-y-3">
   387→            <ColorControl
   388→              label="Color primario (encabezados, acentos)"
   389→              value={design.primaryColor || '#0a3143'}
   390→              onChange={(v) => update({ primaryColor: v, accentColor: v })}
   391→            />
   392→            <ColorControl
   393→              label="Color de acento"
   394→              value={design.accentColor || design.primaryColor || '#0a3143'}
   395→              onChange={(v) => update({ accentColor: v })}
   396→            />
   397→            <ColorControl
   398→              label="Color del texto"
   399→              value={design.textColor || '#111111'}
   400→              onChange={(v) => update({ textColor: v })}
   401→            />
   402→            <ColorControl
   403→              label="Color de fondo"
   404→              value={design.backgroundColor || '#ffffff'}
   405→              onChange={(v) => update({ backgroundColor: v })}
   406→            />
   407→            <button
   408→              type="button"
   409→              onClick={() => update({ primaryColor: '#0a3143', accentColor: '#0a3143', textColor: '#111111', backgroundColor: '#ffffff' })}
   410→              className="text-[10px] text-muted-foreground underline hover:text-foreground"
   411→            >
   412→              Restablecer colores CENPOD
   413→            </button>
   414→          </CardContent>
   415→        </Card>
   416→
   417→        {/* Tipografía */}
   418→        <Card>
   419→          <CardHeader>
   420→            <CardTitle className="text-base">Tipografía</CardTitle>
   421→          </CardHeader>
   422→          <CardContent className="space-y-3">
   423→            <div>
   424→              <Label className="text-xs">Familia tipográfica</Label>
   425→              <Select
   426→                value={design.fontFamilyCategory || 'serif'}
   427→                onValueChange={(v) => {
   428→                  const cat = v as 'serif' | 'sans-serif' | 'system'
   429→                  const fam = cat === 'serif'
   430→                    ? "'Times New Roman', Georgia, serif"
   431→                    : cat === 'sans-serif'
   432→                      ? "Arial, Helvetica, sans-serif"
   433→                      : "ui-sans-serif, system-ui, -apple-system, 'Segoe UI', Roboto, sans-serif"
   434→                  update({ fontFamilyCategory: cat, fontFamily: fam })
   435→                }}
   436→              >
   437→                <SelectTrigger><SelectValue /></SelectTrigger>
   438→                <SelectContent>
   439→                  <SelectItem value="serif">Serif (Times / Georgia)</SelectItem>
   440→                  <SelectItem value="sans-serif">Sans-serif (Arial / Helvetica)</SelectItem>
   441→                  <SelectItem value="system">System (UI)</SelectItem>
   442→                </SelectContent>
   443→              </Select>
   444→            </div>
   445→            <SliderControl
   446→              label="Tamaño de fuente"
   447→              value={design.fontSize ?? 13}
   448→              min={10}
   449→              max={18}
   450→              unit="px"
   451→              onChange={(v) => update({ fontSize: v })}
   452→            />
   453→            <SliderControl
   454→              label="Interlineado"
   455→              value={Math.round((design.lineHeight ?? 1.5) * 10) / 10}
   456→              min={12}
   457→              max={20}
   458→              step={1}
   459→              unit=""
   460→              transform={(v) => v / 10}
   461→              onChange={(v) => update({ lineHeight: Math.round(v * 10) / 10 })}
   462→            />
   463→          </CardContent>
   464→        </Card>
   465→
   466→        {/* Layout */}
   467→        <Card>
   468→          <CardHeader>
   469→            <CardTitle className="text-base">Layout</CardTitle>
   470→          </CardHeader>
   471→          <CardContent className="space-y-3">
   472→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   473→              <ToggleRow label="Encabezado (datos clínica)" value={design.showHeader !== false} onChange={(v) => update({ showHeader: v })} />
   474→              <ToggleRow label="Pie de página" value={design.showFooter !== false} onChange={(v) => update({ showFooter: v })} />
   475→              <ToggleRow label="Símbolo ℞" value={design.showRxSymbol !== false} onChange={(v) => update({ showRxSymbol: v })} />
   476→              <ToggleRow label="Info del paciente" value={design.showPatientInfo !== false} onChange={(v) => update({ showPatientInfo: v })} />
   477→              <ToggleRow label="Info del doctor" value={design.showDoctorInfo !== false} onChange={(v) => update({ showDoctorInfo: v })} />
   478→              <ToggleRow label="Diagnóstico" value={design.showDiagnosis !== false} onChange={(v) => update({ showDiagnosis: v })} />
   479→              <ToggleRow label="Tabla de medicamentos o productos" value={design.showMedications !== false} onChange={(v) => update({ showMedications: v })} />
   480→              <ToggleRow label="Indicaciones" value={design.showIndications !== false} onChange={(v) => update({ showIndications: v })} />
   481→              <ToggleRow label="Línea de firma" value={design.showSignature !== false} onChange={(v) => update({ showSignature: v })} />
   482→            </div>
   483→            <Separator />
   484→            <div className="space-y-3">
   485→              <div className="space-y-1">
   486→                <Label className="text-xs">Nombre del profesional en la receta</Label>
   487→                <Select
   488→                  value={design.doctorNameMode || 'podologist'}
   489→                  onValueChange={(v) => update({ doctorNameMode: v as any })}
   490→                >
   491→                  <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   492→                  <SelectContent>
   493→                    <SelectItem value="podologist">Nombre del podólogo que receta</SelectItem>
   494→                    <SelectItem value="fixed">Nombre fijo (responsable de la clínica)</SelectItem>
   495→                  </SelectContent>
   496→                </Select>
   497→                {design.doctorNameMode === 'fixed' && (
   498→                  <Input
   499→                    value={design.doctorFixedName || ''}
   500→                    onChange={(e) => update({ doctorFixedName: e.target.value })}
   501→                    placeholder="Nombre del responsable"
   502→                    className="h-9 mt-1"
   503→                  />
   504→                )}
   505→                <p className="text-[10px] text-muted-foreground">
   506→                  La cédula profesional siempre es la misma (se configura en Equipo → Podólogo).
   507→                </p>
   508→              </div>
   509→              <Separator />
   510→              <div className="space-y-1">
   511→                <Label className="text-xs">Texto bajo la firma (etiqueta)</Label>
   512→                <Input
   513→                  value={design.signatureLabel || ''}
   514→                  onChange={(e) => update({ signatureLabel: e.target.value })}
   515→                  placeholder="Cédula profesional"
   516→                  className="h-9"
   517→                />
   518→                <p className="text-[10px] text-muted-foreground">
   519→                  Ejemplos: &ldquo;Cédula profesional&rdquo;, &ldquo;Número de certificación&rdquo;, &ldquo;Matrícula&rdquo;
   520→                </p>
   521→              </div>
   522→            </div>
   523→          </CardContent>
   524→        </Card>
   525→
   526→        {/* Acciones */}
   527→        <div className="flex flex-wrap gap-2 justify-end">
   528→          <Button variant="outline" onClick={handleReset}>
   529→            <RotateCcw className="h-4 w-4" /> Restablecer
   530→          </Button>
   531→          <Button variant="outline" onClick={handlePrintTest}>
   532→            <Printer className="h-4 w-4" /> Imprimir prueba
   533→          </Button>
   534→          <Button
   535→            onClick={() => saveMut.mutate(design)}
   536→            disabled={saveMut.isPending}
