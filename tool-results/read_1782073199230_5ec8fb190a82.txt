    92→    92→export function PrescriptionEditor() {
    93→    93→  const qc = useQueryClient()
    94→    94→  const fileInputRef = useRef<HTMLInputElement>(null)
    95→    95→
    96→    96→  // Cargar configuración existente
    97→    97→  const { data: configData, isLoading } = useQuery<any>({
    98→    98→    queryKey: ['config'],
    99→    99→    queryFn: () => fetch('/api/config').then((r) => r.json()),
   100→   100→  })
   101→   101→
   102→   102→  const clinic = configData?.clinic
   103→   103→
   104→   104→  // Estado del diseño. Se inicializa una sola vez cuando llega la data.
   105→   105→  const [design, setDesign] = useState<PrescriptionDesign | null>(null)
   106→   106→  const [uploadedLogoUrl, setUploadedLogoUrl] = useState<string | null>(null)
   107→   107→
   108→   108→  // Inicializar diseño desde config (una sola vez)
   109→   109→  if (!isLoading && configData && !design) {
   110→   110→    let parsed: PrescriptionDesign = DEFAULT_DESIGN
   111→   111→    const raw = configData?.config?.prescriptionDesign
   112→   112→    if (raw) {
   113→   113→      try {
   114→   114→        parsed = { ...DEFAULT_DESIGN, ...(JSON.parse(raw) as Partial<PrescriptionDesign>) }
   115→   115→      } catch {
   116→   116→        parsed = DEFAULT_DESIGN
   117→   117→      }
   118→   118→    }
   119→   119→    // Si la clínica ya tiene logo, mostrarlo por defecto (logoUrl='auto')
   120→   120→    if (clinic?.logoUrl && parsed.logoUrl === undefined) {
   121→   121→      parsed.logoUrl = 'auto'
   122→   122→    }
   123→   123→    setDesign(parsed)
   124→   124→  }
   125→   125→
   126→   126→  function update(patch: Partial<PrescriptionDesign>) {
   127→   127→    setDesign((d) => (d ? { ...d, ...patch } : d))
   128→   128→  }
   129→   129→
   130→   130→  // Guardar diseño
   131→   131→  const saveMut = useMutation({
   132→   132→    mutationFn: async (body: PrescriptionDesign) => {
   133→   133→      const r = await fetch('/api/config/plantillas', {
   134→   134→        method: 'PATCH',
   135→   135→        headers: { 'Content-Type': 'application/json' },
   136→   136→        body: JSON.stringify({ prescriptionDesign: JSON.stringify(body) }),
   137→   137→      })
   138→   138→      if (!r.ok) {
   139→   139→        const e = await r.json().catch(() => ({}))
   140→   140→        throw new Error(e.error || 'Error al guardar')
   141→   141→      }
   142→   142→      return r.json()
   143→   143→    },
   144→   144→    onSuccess: () => {
   145→   145→      toast.success('Diseño de receta guardado')
   146→   146→      qc.invalidateQueries({ queryKey: ['config'] })
   147→   147→    },
   148→   148→    onError: (e: Error) => toast.error(e.message),
   149→   149→  })
   150→   150→
   151→   151→  // Subir logo
   152→   152→  const uploadMut = useMutation({
   153→   153→    mutationFn: async (file: File) => {
   154→   154→      const fd = new FormData()
   155→   155→      fd.append('file', file)
   156→   156→      const r = await fetch('/api/config/logo', { method: 'POST', body: fd })
   157→   157→      if (!r.ok) {
   158→   158→        const e = await r.json().catch(() => ({}))
   159→   159→        throw new Error(e.error || 'Error al subir el logo')
   160→   160→      }
   161→   161→      return r.json() as Promise<{ url: string }>
   162→   162→    },
   163→   163→    onSuccess: (data) => {
   164→   164→      toast.success('Logo subido correctamente')
   165→   165→      setUploadedLogoUrl(data.url)
   166→   166→      update({ logoUrl: data.url })
   167→   167→      qc.invalidateQueries({ queryKey: ['config'] })
   168→   168→    },
   169→   169→    onError: (e: Error) => toast.error(e.message),
   170→   170→  })
   171→   171→
   172→   172→  function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
   173→   173→    const f = e.target.files?.[0]
   174→   174→    if (!f) return
   175→   175→    uploadMut.mutate(f)
   176→   176→  }

   180→   180→    setDesign({ ...DEFAULT_DESIGN, logoUrl: clinic?.logoUrl ? 'auto' : 'none' })
   181→   181→    toast.info('Diseño restablecido. Recuerda guardar.')
   182→   182→  }
