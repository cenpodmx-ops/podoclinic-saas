   293→function StatusCard({
   294→  op,
   295→  onAbrir,
   296→  onCerrar,
   297→  saving,
   298→}: {
   299→  op: OperacionesResponse
   300→  onAbrir: (openingFund: number, notes?: string) => void
   301→  onCerrar: () => void
   302→  saving: boolean
   303→}) {
   304→  const [openingFund, setOpeningFund] = useState('0')
   305→  const [notes, setNotes] = useState('')
   306→
   307→  if (op.status === 'CERRADA_SIN_ABRIR') {
   308→    return (
   309→      <Card className="shadow-sm border-2 border-dashed">
   310→        <CardHeader className="pb-3">
   311→          <CardTitle className="text-base flex items-center gap-2">
   312→            <Unlock className="h-5 w-5 text-amber-600" />
   313→            Sucursal cerrada · abre para registrar operaciones
   314→          </CardTitle>
   315→        </CardHeader>
   316→        <CardContent className="space-y-3">
   317→          <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
   318→            <div>
   319→              <Label className="text-xs">Fondo de apertura (MXN)</Label>
   320→              <Input
   321→                type="number"
   322→                min="0"
   323→                step="0.01"
   324→                value={openingFund}
   325→                onChange={(e) => setOpeningFund(e.target.value)}
   326→                className="text-lg font-semibold"
   327→              />
   328→              <p className="text-[10px] text-muted-foreground mt-1">
   329→                Efectivo inicial en caja al abrir.
   330→              </p>
   331→            </div>
   332→            <div className="md:col-span-2">
   333→              <Label className="text-xs">Notas de apertura (opcional)</Label>
   334→              <Textarea
   335→                rows={2}
   336→                value={notes}
   337→                onChange={(e) => setNotes(e.target.value)}
   338→                placeholder="Ej. Faltó una llave, retraso de 15 min…"
   339→              />
   340→            </div>
   341→          </div>
   342→          <div className="flex justify-end">
   343→            <Button
   344→              size="lg"
   345→              disabled={saving}
   346→              onClick={() => onAbrir(Number(openingFund) || 0, notes || undefined)}
   347→              style={{ backgroundColor: '#0a3143' }}
   348→            >
   349→              <Unlock className="h-4 w-4 mr-2" /> Abrir sucursal
   350→            </Button>
   351→          </div>
   352→        </CardContent>
   353→      </Card>
   354→    )
   355→  }
   356→
   357→  if (op.status === 'ABIERTA') {
   358→    return (
   359→      <Card className="shadow-sm border-emerald-300 bg-emerald-50/50">
   360→        <CardContent className="p-4 flex flex-col md:flex-row md:items-center justify-between gap-3">
   361→          <div className="flex items-center gap-3">
   362→            <div className="p-3 rounded-full bg-emerald-100">
   363→              <DoorOpen className="h-6 w-6 text-emerald-700" />
   364→            </div>
   365→            <div>
   366→              <h3 className="font-semibold text-emerald-900">Sucursal abierta</h3>
   367→              <p className="text-xs text-emerald-700">
   368→                Abierta por <strong>{op.apertura?.performedBy || '—'}</strong> · Fondo {fmtMoney(op.apertura?.openingFund ?? 0)} · {op.apertura && new Date(new Date(op.apertura.createdAt).getTime() - 7 * 60 * 60 * 1000).toLocaleString('es-MX', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' })}
   369→              </p>
   370→            </div>
   371→          </div>
   372→          <Button
   373→            size="lg"
   374→            variant="destructive"
   375→            onClick={onCerrar}
   376→          >
   377→            <Lock className="h-4 w-4 mr-2" /> Cerrar sucursal
   378→          </Button>
   379→        </CardContent>
   380→      </Card>
   381→    )
   382→  }
   383→
   384→  // CERRADA
   385→  return (
   386→    <Card className="shadow-sm border-slate-300">
   387→      <CardContent className="p-4 flex flex-col md:flex-row md:items-center justify-between gap-3">
   388→        <div className="flex items-center gap-3">
   389→          <div className="p-3 rounded-full bg-slate-200">
   390→            <DoorClosed className="h-6 w-6 text-slate-700" />
   391→          </div>
   392→          <div>
   393→            <h3 className="font-semibold text-slate-900">Sucursal cerrada por hoy</h3>
   394→            <p className="text-xs text-slate-700">
   395→              Cerrada por <strong>{op.cierre?.performedBy || '—'}</strong> · {op.cierre && new Date(new Date(op.cierre.createdAt).getTime() - 7 * 60 * 60 * 1000).toLocaleString('es-MX', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' })}
   396→              {op.cierre && op.cierre.difference !== null && op.cierre.difference !== 0 && (() => {
   397→                const diff = op.cierre!.difference!
   398→                return (
   399→                  <span className={`ml-2 font-semibold ${diff > 0 ? 'text-emerald-700' : 'text-red-700'}`}>
   400→                    Diferencia: {diff > 0 ? '+' : ''}{fmtMoney(diff)}
   401→                  </span>
   402→                )
   403→              })()}
   404→            </p>
   405→          </div>
   406→        </div>
   407→        <div className="flex gap-2">
   408→          <Button
   409→            variant="outline"
   410→            onClick={() => op.cierre && window.open(`/api/operaciones/${op.cierre.id}/pdf`, '_blank')}
   411→          >
   412→            <Printer className="h-4 w-4 mr-1" /> Ver reporte
   413→          </Button>
   414→        </div>
   415→      </CardContent>
   416→    </Card>
   417→  )
   418→}

   711→function WhatsAppDialog({
   712→  open,
   713→  onOpenChange,
   714→  onSend,
   715→}: {
   716→  open: boolean
   717→  onOpenChange: (v: boolean) => void
   718→  onSend: (phone: string) => void
   719→}) {
   720→  const [name, setName] = useState('')
   721→  const [phone, setPhone] = useState('')
   722→  const [contacts, setContacts] = useState<WaContact[]>([])
   723→
   724→  // Cargar contactos guardados al montar
   725→  useEffect(() => {
   726→    try {
   727→      const saved = localStorage.getItem('cenpod_wa_contacts')
   728→      if (saved) {
   729→        const parsed = JSON.parse(saved) as WaContact[]
   730→        // Usar setContacts fuera del effect body para evitar cascada de renders
   731→        setTimeout(() => setContacts(parsed), 0)
   732→      }
   733→    } catch {}
   734→  }, [])
   735→
   736→  function saveContact() {
   737→    const cleaned = phone.replace(/[^0-9]/g, '')
   738→    if (cleaned.length < 10) {
   739→      toast.error('Teléfono inválido (mínimo 10 dígitos)')
   740→      return
   741→    }
   742→    const newContact: WaContact = { name: name || 'Sin nombre', phone: cleaned }
   743→    const updated = [...contacts.filter((c) => c.phone !== cleaned), newContact]
   744→    setContacts(updated)
   745→    localStorage.setItem('cenpod_wa_contacts', JSON.stringify(updated))
   746→    setName('')
   747→    setPhone('')
   748→    toast.success('Contacto guardado')
   749→  }
   750→
   751→  function deleteContact(phone: string) {
   752→    const updated = contacts.filter((c) => c.phone !== phone)
   753→    setContacts(updated)
   754→    localStorage.setItem('cenpod_wa_contacts', JSON.stringify(updated))
   755→  }
   756→
   757→  return (
   758→    <Dialog open={open} onOpenChange={onOpenChange}>
   759→      <DialogContent className="max-w-md">
   760→        <DialogHeader>
   761→          <DialogTitle>Enviar por WhatsApp</DialogTitle>
   762→          <DialogDescription>
   763→            Selecciona un contacto guardado o agrega uno nuevo. Se abrirá WhatsApp con el mensaje del cierre.
   764→          </DialogDescription>
   765→        </DialogHeader>

   767→          {/* Contactos guardados */}
   768→          {contacts.length > 0 && (
   769→            <div className="space-y-1.5">
   770→              <Label className="text-xs font-semibold text-muted-foreground">Contactos guardados</Label>
   771→              {contacts.map((c) => (
   772→                <div key={c.phone} className="flex items-center justify-between rounded-md border p-2">
   773→                  <div className="flex items-center gap-2">
   774→                    <span className="text-sm font-medium">{c.name}</span>
   775→                    <span className="text-xs text-muted-foreground">{c.phone}</span>
   776→                  </div>
   777→                  <div className="flex gap-1">
   778→                    <Button size="sm" variant="ghost" onClick={() => { onSend(c.phone); onOpenChange(false) }}>
   779→                      <Send className="h-3.5 w-3.5" />
   780→                    </Button>
   781→                    <Button size="sm" variant="ghost" className="text-red-600" onClick={() => deleteContact(c.phone)}>
   782→                      <XCircle className="h-3.5 w-3.5" />
   783→                    </Button>
   784→                  </div>
   785→                </div>
   786→              ))}
   787→            </div>
   788→          )}
   789→
   790→          <div className="border-t pt-3 space-y-2">
   791→            <Label className="text-xs font-semibold text-muted-foreground">Agregar nuevo contacto</Label>
   792→            <div className="grid grid-cols-2 gap-2">
   793→              <Input
   794→                placeholder="Nombre"
   795→                value={name}
   796→                onChange={(e) => setName(e.target.value)}
   797→              />
   798→              <Input
   799→                placeholder="Teléfono (10 dígitos)"
   800→                inputMode="tel"
   801→                value={phone}
   802→                onChange={(e) => setPhone(e.target.value)}
   803→              />
   804→            </div>
   805→            <p className="text-xs text-muted-foreground">Se agrega +52 automáticamente si son 10 dígitos.</p>
   806→          </div>
   807→        </div>
   808→        <DialogFooter className="flex gap-2">
   809→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
   810→          {phone.replace(/[^0-9]/g, '').length >= 10 && (
   811→            <Button variant="outline" onClick={saveContact}>
   812→              Guardar contacto
   813→            </Button>
   814→          )}
   815→          <Button
   816→            style={{ backgroundColor: '#25D366' }}
   817→            disabled={phone.replace(/[^0-9]/g, '').length < 10}
   818→            onClick={() => { onSend(phone); onOpenChange(false) }}
   819→          >
   820→            <Send className="h-4 w-4 mr-1" /> Abrir WhatsApp
   821→          </Button>
   822→        </DialogFooter>
   823→      </DialogContent>
   824→    </Dialog>

   872→function CierreDialog({
   873→  open,
   874→  onClose,
   875→  expectedCash,
   876→  onClosed,
   877→}: {
   878→  open: boolean
   879→  onClose: () => void
   880→  expectedCash: number
   881→  onClosed: (cierreId: string) => void
   882→}) {
   883→  const [countedCash, setCountedCash] = useState('')
   884→  const [notes, setNotes] = useState('')
   885→  const [saving, setSaving] = useState(false)
   886→  const sigRef = useRef<SignaturePadHandle>(null)
   887→
   888→  // Reset form on open
   889→  useEffect(() => {
   890→    if (open) {
   891→      setCountedCash('')
   892→      setNotes('')
   893→      sigRef.current?.clear()
   894→    }
   895→  }, [open])
   896→
   897→  const difference = (Number(countedCash) || 0) - expectedCash
   898→  const diffColor = difference === 0 ? 'text-slate-700' : difference > 0 ? 'text-emerald-700' : 'text-red-700'
   899→
   900→  const submit = async () => {
   901→    if (countedCash === '' || isNaN(Number(countedCash))) {
   902→      toast.error('Ingresa el efectivo contado')
   903→      return
   904→    }
   905→    setSaving(true)
   906→    try {
   907→      const signatureData = sigRef.current?.getDataUrl() || null
   908→      const r = await fetch('/api/operaciones/cierre', {
   909→        method: 'POST',
   910→        headers: { 'Content-Type': 'application/json' },
   911→        body: JSON.stringify({
   912→          countedCash: Number(countedCash),
   913→          notes: notes || undefined,
   914→          signatureData,
   915→        }),
   916→      })
   917→      if (!r.ok) {
   918→        const e = await r.json()
   919→        throw new Error(e.error || 'Error al cerrar')
   920→      }
   921→      const data = await r.json()
   922→      toast.success('Sucursal cerrada')
   923→      // Reset
   924→      setCountedCash('')
   925→      setNotes('')
   926→      sigRef.current?.clear()
   927→      onClosed(data.cierre.id)
   928→    } catch (e: any) {
   929→      toast.error(e.message || 'Error al cerrar')
   930→    } finally {
   931→      setSaving(false)
   932→    }
   933→  }
   934→
   935→  return (
   936→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   937→      <DialogContent className="max-w-lg">
   938→        <DialogHeader>
   939→          <DialogTitle className="flex items-center gap-2">
   940→            <Lock className="h-5 w-5 text-red-700" /> Cerrar sucursal
   941→          </DialogTitle>
   942→          <DialogDescription>
   943→            Confirma el efectivo contado, incidencias y firma. Se generará un reporte imprimible.
   944→          </DialogDescription>
   945→        </DialogHeader>
   946→
   947→        <div className="space-y-4">
   948→          <div className="grid grid-cols-2 gap-3">
   949→            <div>
   950→              <Label className="text-xs">Efectivo esperado</Label>
   951→              <div className="px-3 py-2 border rounded-md bg-muted/40 font-semibold" style={{ color: '#0a3143' }}>
   952→                {fmtMoney(expectedCash)}
   953→              </div>
   954→            </div>
   955→            <div>
   956→              <Label className="text-xs">Efectivo contado *</Label>
   957→              <Input
   958→                type="number"
   959→                step="0.01"
   960→                min="0"
   961→                value={countedCash}
   962→                onChange={(e) => setCountedCash(e.target.value)}
   963→                placeholder="0.00"
   964→                className="text-lg font-semibold"
   965→              />
   966→            </div>
   967→          </div>
   968→
   969→          <div className="flex items-center justify-between border rounded-md p-3 bg-muted/30">
   970→            <span className="text-sm font-medium">Diferencia</span>
   971→            <span className={`text-xl font-bold ${diffColor}`}>
   972→              {difference >= 0 ? '+' : ''}{fmtMoney(difference)}
   973→            </span>
   974→          </div>
   975→
   976→          <div>
   977→            <Label className="text-xs">Incidencias / Notas</Label>
   978→            <Textarea
   979→              rows={3}
   980→              value={notes}
   981→              onChange={(e) => setNotes(e.target.value)}
   982→              placeholder="Anomalías, faltantes, observaciones del día…"
   983→            />
   984→          </div>
