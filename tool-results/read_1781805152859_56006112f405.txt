    67→export default function ConsultaPage() {
    68→  const params = useSearchParams()
    69→  const router = useRouter()
    70→  const qc = useQueryClient()
    71→
    72→  const citaId = params.get('cita') || ''
    73→
    74→  const [showTicket, setShowTicket] = useState(false)
    75→  const [savedConsultationId, setSavedConsultationId] = useState<string | null>(null)
    76→  // Override de fase: cuando el usuario inicia/cobra/continúa, se setea
    77→  // manualmente y toma prioridad sobre la derivación automática.
    78→  const [manualPhase, setManualPhase] = useState<Phase | null>(null)
    79→
    80→  // ── Query principal: trae la cita + paciente + consulta existente (si la hay)
    81→  const consultaQ = useQuery<ConsultaApiResponse>({
    82→    queryKey: ['consulta-context', citaId],
    83→    queryFn: async () => {
    84→      const r = await fetch(`/api/consultas?cita=${encodeURIComponent(citaId)}`)
    85→      if (!r.ok) throw new Error('No se pudo cargar la cita')
    86→      return r.json()
    87→    },
    88→    enabled: !!citaId,
    89→    staleTime: 0,
    90→  })
    91→
    92→  // ── Configuración de la clínica (diagnósticos predefinidos + datos para ticket)
    93→  const configQ = useQuery<ConfigResponse>({
    94→    queryKey: ['config-clinica'],
    95→    queryFn: async () => {
    96→      const r = await fetch('/api/config')
    97→      if (!r.ok) throw new Error('No se pudo cargar configuración')
    98→      return r.json()
    99→    },
   100→    staleTime: 60_000,
   101→  })
   102→
   103→  // ── Servicios disponibles (selector en step 2)
   104→  const servicesQ = useQuery<{ rows: ServiceItem[] }>({
   105→    queryKey: ['servicios-list'],
   106→    queryFn: async () => {
   107→      const r = await fetch('/api/servicios')
   108→      if (!r.ok) throw new Error('No se pudo cargar servicios')
   109→      return r.json()
   110→    },
   111→    staleTime: 60_000,
   112→  })
   113→
   114→  // ── Fase derivada automáticamente de los datos
   115→  const autoPhase: Phase = useMemo(() => {
   116→    if (!citaId) return 'list'
   117→    if (consultaQ.isLoading || !consultaQ.data) return 'loading'
   118→    if (consultaQ.isError) return 'list'
   119→    const { appointment, consultation } = consultaQ.data
   120→    if (consultation && consultation.paid) return 'finalized'
   121→    if (consultation && !consultation.paid) return 'saved-unpaid'
   122→    if (appointment.status === 'EN_CONSULTA') return 'form'
   123→    if (appointment.status === 'FINALIZADA') return 'finalized'
   124→    return 'confirm-start'
   125→  }, [citaId, consultaQ.isLoading, consultaQ.isError, consultaQ.data])
   126→
   127→  // El override manual tiene prioridad; se reinicia cuando cambia la cita
   128→  const phase: Phase = useMemo(() => {
   129→    if (manualPhase) {
   130→      // Si la data cargó y refleja un estado "final", respetar la reality
   131→      if (consultaQ.data?.consultation?.paid) {
   132→        // Pero si el usuario acaba de cobrar y estamos en 'success', mostrar success
   133→        if (manualPhase === 'success') return 'success'
   134→        return 'finalized'
   135→      }
   136→      return manualPhase
   137→    }
   138→    return autoPhase
   139→  }, [manualPhase, autoPhase, consultaQ.data])
   140→
   141→  // Reset del override cuando cambia la cita (URL)
   142→  const [lastCita, setLastCita] = useState(citaId)
   143→  if (citaId !== lastCita) {
   144→    setLastCita(citaId)
   145→    setManualPhase(null)
   146→    setSavedConsultationId(null)
   147→  }
   148→
   149→  // ── Mutación para iniciar consulta (PATCH cita → EN_CONSULTA)
   150→  const startMut = useMutation({
   151→    mutationFn: async () => {
   152→      const r = await fetch(`/api/citas/${citaId}`, {
   153→        method: 'PATCH',
   154→        headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify({ status: 'EN_CONSULTA' }),
   156→      })
   157→      if (!r.ok) {
   158→        const e = await r.json().catch(() => ({}))
   159→        throw new Error(e.error || 'No se pudo iniciar la consulta')
   160→      }
   161→      return r.json()
   162→    },
   163→    onSuccess: () => {

   165→      qc.invalidateQueries({ queryKey: ['consulta-context', citaId] })
   166→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   167→      setManualPhase('form')
   168→    },
   169→    onError: (e: Error) => toast.error(e.message),
   170→  })
   171→
   172→  // ──────────────────────────────────────────────────────────
   173→  //  RENDER
   174→  // ──────────────────────────────────────────────────────────
   175→  return (
   176→    <div className="p-4 md:p-6 space-y-4 max-w-[1100px] mx-auto">
   177→      <div className="flex items-center gap-3">
   178→        <Button variant="ghost" size="icon" onClick={() => router.push('/agenda')}>
   179→          <ArrowLeft className="h-4 w-4" />
   180→        </Button>
   181→        <div>
   182→          <h1 className="text-xl font-bold flex items-center gap-2">
   183→            <Stethoscope className="h-5 w-5" style={{ color: '#0a3143' }} />
   184→            Consulta
   185→          </h1>
   186→          <p className="text-xs text-muted-foreground">Módulo 02 · Registro clínico y cobro</p>
   187→        </div>
   188→      </div>
   189→
   190→      {phase === 'loading' && <LoadingSkeleton />}
   191→
   192→      {phase === 'list' && <ConsultasList onPick={(id) => router.push(`/consulta?cita=${id}`)} />}
   193→
   194→      {phase === 'confirm-start' && consultaQ.data && (
   195→        <ConfirmStart
   196→          data={consultaQ.data}
   197→          onConfirm={() => startMut.mutate()}
   198→          loading={startMut.isPending}
   199→        />
   200→      )}
   201→
   202→      {phase === 'form' && consultaQ.data && (
   203→        <ConsultaForm
   204→          key={citaId}
   205→          citaId={citaId}
   206→          data={consultaQ.data}
   207→          config={configQ.data}
   208→          services={servicesQ.data?.rows || []}
   209→          onSuccess={(id, paid) => {
   210→            setSavedConsultationId(id)
   211→            qc.invalidateQueries({ queryKey: ['consulta-context', citaId] })
   212→            qc.invalidateQueries({ queryKey: ['dashboard'] })
   213→            if (paid) {
   214→              setManualPhase('success')
   215→              setShowTicket(true)
   216→            } else {
   217→              setManualPhase('saved-unpaid')
   218→            }
   219→          }}
   220→          onCancel={() => router.push('/agenda')}
   221→        />
   222→      )}
   223→
   224→      {phase === 'saved-unpaid' && consultaQ.data && (
   225→        <SavedUnpaidView
   226→          data={consultaQ.data}
   227→          config={configQ.data}
   228→          consultationId={savedConsultationId}
   229→          onContinue={() => setManualPhase('form')}
   230→          onTicket={() => setShowTicket(true)}
   231→        />
   232→      )}
   233→
   234→      {phase === 'finalized' && consultaQ.data && (
   235→        <FinalizedView
   236→          data={consultaQ.data}
   237→          config={configQ.data}
   238→          onTicket={() => setShowTicket(true)}
   239→        />
   240→      )}
   241→
   242→      {phase === 'success' && consultaQ.data && (
   243→        <SuccessView
   244→          data={consultaQ.data}
   245→          config={configQ.data}
   246→          onTicket={() => setShowTicket(true)}
   247→        />
   248→      )}
   249→
   250→      {/* Modal de ticket */}
   251→      {showTicket && consultaQ.data && (
   252→        <TicketDialog
   253→          open={showTicket}
   254→          onOpenChange={setShowTicket}
   255→          data={consultaQ.data}
   256→          config={configQ.data}
   257→          consultationId={savedConsultationId}
   258→        />
   259→      )}
   260→    </div>
   261→  )
   262→}

   452→        </CardContent>

   454→    </div>

   468→//  FORMULARIO DE CONSULTA — 3 PASOS

   470→function ConsultaForm({
   471→  citaId,
   472→  data,
   473→  config,
   474→  services,
   475→  onSuccess,
   476→  onCancel,
   477→}: {
   478→  citaId: string
   479→  data: ConsultaApiResponse
   480→  config: ConfigResponse | undefined
   481→  services: ServiceItem[]
   482→  onSuccess: (id: string, paid: boolean) => void
   483→  onCancel: () => void
   484→}) {
   485→  const qc = useQueryClient()
   486→  const { appointment, patient, podologist, consultation } = data
   487→
   488→  const [step, setStep] = useState<1 | 2 | 3>(1)
   489→  const [saving, setSaving] = useState(false)
   490→
   491→  // Step 1 fields
   492→  const [reason, setReason] = useState(consultation?.reason || appointment.reason || '')
   493→  const [referredBy, setReferredBy] = useState<string>(consultation?.referredBy || 'NADIE')
   494→  const [diagnosis, setDiagnosis] = useState(consultation?.diagnosis || '')
   495→  const [treatment, setTreatment] = useState(consultation?.treatment || '')
   496→  const [notes, setNotes] = useState(consultation?.notes || '')
   497→
   498→  // Step 2 fields
   499→  const [items, setItems] = useState<ConsultaItem[]>(
   500→    consultation?.items?.filter((i) => i.type !== 'SERVICIO') || []
   501→  )
   502→  const [consultPrice, setConsultPrice] = useState<number>(
   503→    consultation?.consultPrice ?? appointment.price ?? 0
   504→  )
   505→  const [selectedServiceId, setSelectedServiceId] = useState<string>(appointment.serviceId || '')
   506→  const [discount, setDiscount] = useState<number>(consultation?.discount || 0)
   507→  const [paymentMethod, setPaymentMethod] = useState<string>(
   508→    consultation?.paymentMethod || 'EFECTIVO'
   509→  )
   510→  const [followUpDays, setFollowUpDays] = useState<number | ''>(consultation?.followUpDays ?? '')
   511→
   512→  // Cálculos
   513→  const productsTotal = useMemo(
   514→    () => items.reduce((s, i) => s + i.qty * i.price, 0),
   515→    [items]
   516→  )
   517→  const subtotal = consultPrice + productsTotal
   518→  const total = Math.max(0, subtotal - discount)
   519→
   520→  // Cuando se selecciona un servicio desde el selector, setear precio y agregar item SERVICIO
   521→  function onServiceChange(serviceId: string) {
   522→    setSelectedServiceId(serviceId)
   523→    const svc = services.find((s) => s.id === serviceId)
   524→    if (svc) {
   525→      setConsultPrice(svc.price)
   526→    }
   527→  }
   528→
   529→  function canAdvance(stepN: 1 | 2 | 3) {
   530→    if (stepN === 1) {
   531→      // Validar motivo o diagnóstico (al menos uno con contenido)
   532→      return reason.trim().length > 0 || diagnosis.trim().length > 0
   533→    }
   534→    if (stepN === 2) {
   535→      return consultPrice >= 0 && !!paymentMethod
   536→    }
   537→    return true
   538→  }
   539→
   540→  // ── Mutación: guardar (con o sin cobro)
   541→  const saveMut = useMutation({
   542→    mutationFn: async (paid: boolean) => {
   543→      setSaving(true)
   544→      const payload = {
   545→        appointmentId: citaId,
   546→        reason: reason.trim() || null,
   547→        referredBy,
   548→        diagnosis: diagnosis.trim() || null,
   549→        treatment: treatment.trim() || null,
   550→        notes: notes.trim() || null,

   983→function Stepper({ step }: { step: 1 | 2 | 3 }) {
   984→  const steps = [
   985→    { n: 1, label: 'Datos', icon: ClipboardList },
   986→    { n: 2, label: 'Cobro', icon: CircleDollarSign },
   987→    { n: 3, label: 'Confirmar', icon: ClipboardCheck },
   988→  ] as const
   989→  return (
   990→    <div className="flex items-center gap-1 sm:gap-2">
   991→      {steps.map((s, i) => {
   992→        const Icon = s.icon
   993→        const active = step === s.n
   994→        const done = step > s.n
   995→        return (
   996→          <div key={s.n} className="flex items-center flex-1">
   997→            <div
   998→              className={`flex items-center gap-1.5 px-2.5 py-1.5 rounded-md flex-1 ${
   999→                active ? 'text-white' : done ? 'text-emerald-700 bg-emerald-50' : 'text-muted-foreground bg-muted/50'
  1000→              }`}
  1001→              style={active ? { backgroundColor: '#0a3143' } : undefined}
  1002→            >
  1003→              <Icon className="h-3.5 w-3.5" />
  1004→              <span className="text-xs font-medium hidden sm:inline">{s.label}</span>
  1005→              {done && <Check className="h-3 w-3 ml-auto" />}
  1006→            </div>
  1007→            {i < steps.length - 1 && <ChevronRight className="h-3 w-3 text-muted-foreground shrink-0 mx-0.5" />}
  1008→          </div>
  1009→        )
  1010→      })}
  1011→    </div>
  1012→  )
  1013→}

  1213→function TicketDialog({
  1214→  open,
  1215→  onOpenChange,
  1216→  data,
  1217→  config,
  1218→  consultationId,
  1219→}: {
  1220→  open: boolean
  1221→  onOpenChange: (v: boolean) => void
  1222→  data: ConsultaApiResponse
  1223→  config: ConfigResponse | undefined
  1224→  consultationId: string | null
  1225→}) {
  1226→  const { patient, podologist, consultation } = data
  1227→  if (!consultation) return null
  1228→
  1229→  function handlePrint() {
  1230→    window.print()
  1231→  }
  1232→
  1233→  // Marcar ticket como impreso (no crítico)
  1234→  async function markPrinted() {
  1235→    if (!consultationId || !consultation) return
  1236→    try {
  1237→      await fetch(`/api/consultas/${consultationId}`, {
  1238→        method: 'PATCH',
  1239→        headers: { 'Content-Type': 'application/json' },
  1240→        body: JSON.stringify({ ticketPrinted: true }),
  1241→      })
  1242→    } catch {}
  1243→  }
  1244→
  1245→  return (

  1250→          <DialogDescription>Vista previa del ticket imprimible.</DialogDescription>
  1251→        </DialogHeader>
  1252→        <div className="bg-white p-3 flex justify-center">
  1253→          <TicketPreview
  1254→            data={{
  1255→              clinic: config?.clinic || null,
  1256→              date: new Date(consultation.date),
  1257→              patientName: `${patient.firstName} ${patient.lastName}`,
  1258→              expNumber: patient.expNumber,
  1259→              podologistName: podologist?.name || '—',
  1260→              items: consultation.items.filter((i) => i.type !== 'SERVICIO'),
  1261→              consultPrice: consultation.consultPrice,
  1262→              productsTotal: consultation.productsTotal,
  1263→              discount: consultation.discount,
  1264→              total: consultation.total,
  1265→              paymentMethod: consultation.paymentMethod ? METHOD_LABELS[consultation.paymentMethod] || consultation.paymentMethod : null,
  1266→              followUpDays: consultation.followUpDays,

  1269→        </div>
  1270→        <DialogFooter className="px-3 pb-3 gap-2 sm:gap-2 no-print">

  1272→            Cerrar

  1274→          <Button
  1275→            onClick={() => {
  1276→              markPrinted()
  1277→              handlePrint()
  1278→            }}
  1279→            className="flex-1"
  1280→            style={{ backgroundColor: '#0a3143' }}
  1281→          >
  1282→            <Printer className="h-4 w-4 mr-1" /> Imprimir
  1283→          </Button>
  1284→        </DialogFooter>
  1285→      </DialogContent>
  1286→    </Dialog>
  1287→  )
  1288→}
